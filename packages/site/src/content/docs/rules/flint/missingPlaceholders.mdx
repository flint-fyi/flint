import { Tabs, TabItem } from "@astrojs/starlight/components";

<RuleSummary plugin="flint" rule="missingPlaceholders" />

When reporting linting issues, it's common to include dynamic data in the message using placeholders.

## Examples

<Tabs>
<TabItem label="❌ Incorrect">

```ts
// @noErrors -- simplified rule definition omits required fields like 'about'
import { RuleCreator } from "@flint.fyi/core";
import { typescriptLanguage } from "@flint.fyi/typescript-language";
const ruleCreator = new RuleCreator({
	docs: () => "",
	pluginId: "flint",
	presets: [] as string[],
});
const _ = typescriptLanguage;
interface RuleContext {
	report(options: Record<string, unknown>): void;
}
// ---cut---
ruleCreator.createRule(_, {
	messages: {
		messagesId: {
			primary: "This is a message with a {{placeholder}}",
			secondary: [""],
			suggestions: [""],
		},
	},
	setup(context: RuleContext) {
		context.report({
			message: "messagesId",
		});
	},
});
```

</TabItem>
<TabItem label="✅ Correct">

```ts
// @noErrors -- simplified rule definition omits required fields like 'about'
import { RuleCreator } from "@flint.fyi/core";
import { typescriptLanguage } from "@flint.fyi/typescript-language";
const ruleCreator = new RuleCreator({
	docs: () => "",
	pluginId: "flint",
	presets: [] as string[],
});
const _ = typescriptLanguage;
interface RuleContext {
	report(options: Record<string, unknown>): void;
}
// ---cut---
ruleCreator.createRule(_, {
	messages: {
		messagesId: {
			primary: "This is a message with a {{placeholder}}",
			secondary: [""],
			suggestions: [""],
		},
	},
	setup(context: RuleContext) {
		context.report({
			data: {
				placeholder: "value",
			},
			message: "messagesId",
		});
	},
});
```

</TabItem>
</Tabs>

## Options

This rule is not configurable.
